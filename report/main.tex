\@itempenalty=10000
\@beginparpenalty=10000
\newcount\dl@listid
\newcount\dl@itemno
\newcount\dl@total
\newif\ifdl@inlist
\newif\ifdl@free
\def\dl@setn#1#2{\expandafter\gdef\csname dl@n@#1\endcsname{#2}}
\def\dl@startlist{%
  \global\advance\dl@listid\@ne
  \edef\dl@thisid{\the\dl@listid}%
  \dl@itemno=\z@ \dl@inlisttrue
  \@ifundefined{dl@n@\dl@thisid}{\dl@total=\z@}{\dl@total=\csname dl@n@\dl@thisid\endcsname\relax}}
\def\dl@endlist{\immediate\write\@auxout{\string\dl@setn{\dl@thisid}{\the\dl@itemno}}}
\def\dl@setitempenalty{%
  \ifdl@inlist
    \advance\dl@itemno\@ne
    \ifdl@free \@itempenalty=-\@lowpenalty
    \else\ifnum\dl@total<5 \@itempenalty=\@M
    \else\ifnum\dl@itemno=\tw@ \@itempenalty=\@M
    \else\ifnum\dl@itemno=\dl@total \@itempenalty=\@M
    \else \@itempenalty=-\@lowpenalty
    \fi\fi\fi\fi
  \fi}
\let\dl@orig@item\@item
\def\@item{\dl@setitempenalty\dl@orig@item}
\AtBeginEnvironment{enumerate}{\dl@startlist}
\AtBeginEnvironment{itemize}{\dl@startlist}
\AtEndEnvironment{enumerate}{\dl@endlist}
\AtEndEnvironment{itemize}{\dl@endlist}
\newcommand{\listbreaksallowed}{\dl@freetrue\@itempenalty=-\@lowpenalty}
\AtBeginBibliography{\listbreaksallowed}
