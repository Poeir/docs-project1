\newcommand{\fillin}[1]{{\bfseries\color{red}[#1]}}

% -------------------------------------------------------------
% ภาพหน้าจอ: \appscreen[ความกว้าง]{ชื่อไฟล์ไม่รวมนามสกุล}{คำอธิบายภาพ}
% วางไฟล์ภาพจริงที่ images/screens/<ชื่อไฟล์>.png (หรือ .jpg) ระบบจะแทนกรอบว่างให้อัตโนมัติ
% (ชื่อ \screenshot ใช้ไม่ได้ เพราะ cskkumanual มี environment screenshot สำหรับภาพพร้อมหมายเลขกำกับอยู่แล้ว)
% -------------------------------------------------------------
\newcommand{\appscreen}[3][0.85]{%
  \begin{figure}[H]
    \centering
    \setlength{\fboxsep}{0pt}%
    \IfFileExists{images/screens/#2.png}{%
      \fbox{\includegraphics[width=#1\textwidth]{images/screens/#2.png}}%
    }{\IfFileExists{images/screens/#2.jpg}{%
      \fbox{\includegraphics[width=#1\textwidth]{images/screens/#2.jpg}}%
    }{%
      \fbox{\parbox[c][5.5cm][c]{#1\textwidth}{\centering\fillin{ภาพหน้าจอ: images/screens/#2.png}}}%
    }}
    \caption{#3}
    \label{fig:#2}
  \end{figure}
}
